60template <typename SpiType>
61bool SmartDisplay<SpiType>::sendCommand(uint8_t cmd, const uint8_t* data, uint8_t len) noexcept {
62 if (!bus_->BeginCommand()) {
63 setError(Error::SpiWrite);
64 return false;
65 }
66
67 if (!bus_->WriteBytes(&cmd, 1)) {
68 bus_->EndTransaction();
69 setError(Error::SpiWrite);
70 return false;
71 }
72
73 if (len > 0 && data != nullptr) {
74 if (!bus_->WriteBytes(data, len)) {
75 bus_->EndTransaction();
76 setError(Error::SpiWrite);
77 return false;
78 }
79 }
80
81 bus_->EndTransaction();
82 return true;
83}

95template <typename SpiType>
96bool SmartDisplay<SpiType>::runInitTable() noexcept {
97 std::size_t count = 0;
98 const ssd1331::InitStep* table = ssd1331::GetInitTable(variant_, count);
99
100 for (std::size_t i = 0; i < count; ++i) {
101 if (!sendCommand(table[i].command, table[i].data, table[i].data_len)) {
102 return false;
103 }
104 }
105 return true;

108template <typename SpiType>
109bool SmartDisplay<SpiType>::Initialize(bool enable_vcc) noexcept {
110 if (bus_ == nullptr) {
111 setError(Error::InvalidParam);
112 return false;
113 }
114
115 if (!bus_->EnsureInitialized()) {
116 setError(Error::NotInitialized);
117 return false;
118 }
119
120 bus_->GpioSetInactive(CtrlPin::ChipSelect);
121 bus_->GpioSetActive(CtrlPin::DataCommand);
122 bus_->GpioSetInactive(CtrlPin::VccEnable);
123
124 pulseReset();
125
126 if (enable_vcc) {
127 bus_->GpioSetActive(CtrlPin::VccEnable);
128 bus_->DelayUs(1000);
129 }
130
131 if (!runInitTable()) {
132 return false;
133 }
134
135 if (!SetFullWindow()) {
136 return false;
137 }
138
139 if (config_.graphics == GraphicsBackend::HardwareDirect) {
140 GraphicsContext gfx = CreateGraphicsContext();
141 ConfigurePresenter(gfx);
142 }
143
144 initialized_ = true;
145 return true;
146}

178template <typename SpiType>
179bool SmartDisplay<SpiType>::SetWindow(const Rect& rect) noexcept {
180 if (rect.x1 < rect.x0 || rect.y1 < rect.y0 || rect.x1 >= geometry_.width ||
181 rect.y1 >= geometry_.height) {
182 setError(Error::OutOfRange);
183 return false;
184 }
185
186 const uint8_t col_data[2] = {rect.x0, rect.x1};
187 const uint8_t row_data[2] = {rect.y0, rect.y1};
188
189 if (!sendCommand(static_cast<uint8_t>(ssd1331::Command::SetColumnAddress), col_data, 2)) {
190 return false;
191 }
192 if (!sendCommand(static_cast<uint8_t>(ssd1331::Command::SetRowAddress), row_data, 2)) {
193 return false;
194 }
195
196 window_ = rect;
197 return true;
198}

206template <typename SpiType>
207bool SmartDisplay<SpiType>::WritePixels(const uint8_t* data, std::size_t byte_len) noexcept {
208 if (!initialized_) {
209 setError(Error::NotInitialized);
210 return false;
211 }
212 if (data == nullptr || byte_len == 0 || (byte_len % 2) != 0) {
213 setError(Error::InvalidParam);
214 return false;
215 }
216
217 if (!bus_->BeginData()) {
218 setError(Error::SpiWrite);
219 return false;
220 }
221
222 if (!bus_->WriteBytes(data, byte_len)) {
223 bus_->EndTransaction();
224 setError(Error::SpiWrite);
225 return false;
226 }
227
228 bus_->EndTransaction();
229 return true;
230}

285template <typename SpiType>
286void SmartDisplay<SpiType>::colorToSsd1331(Color565 color, uint8_t out[3]) noexcept {
287 out[0] = static_cast<uint8_t>((color >> 11) & 0x1F); // A: red 5-bit
288 out[1] = static_cast<uint8_t>((color >> 5) & 0x3F); // B: green 6-bit
289 out[2] = static_cast<uint8_t>(color & 0x1F); // C: blue 5-bit
290}

292template <typename SpiType>
293bool SmartDisplay<SpiType>::DrawLine(const Point& start, const Point& end, Color565 color) noexcept {
294 uint8_t rgb[3];
295 colorToSsd1331(color, rgb);
296 const uint8_t data[7] = {start.x, start.y, end.x, end.y,
297 static_cast<uint8_t>(rgb[2] << 1), rgb[1],
298 static_cast<uint8_t>(rgb[0] << 1)};
299 return sendCommand(static_cast<uint8_t>(ssd1331::Command::DrawLine), data, sizeof(data));

302template <typename SpiType>
303bool SmartDisplay<SpiType>::DrawRectangle(const Rect& rect, Color565 color, bool filled) noexcept {
304 if (filled) {
305 const uint8_t fill_cfg[2] = {0x01, 0x00};
306 if (!sendCommand(static_cast<uint8_t>(ssd1331::Command::FillEnable), fill_cfg, 2)) {
307 return false;
308 }
309 }
310
311 uint8_t rgb[3];
312 colorToSsd1331(color, rgb);
313 const uint8_t data[10] = {rect.x0, rect.y0, rect.x1, rect.y1,
314 static_cast<uint8_t>(rgb[2] << 1), rgb[1],
315 static_cast<uint8_t>(rgb[0] << 1),
316 static_cast<uint8_t>(rgb[2] << 1), rgb[1],
317 static_cast<uint8_t>(rgb[0] << 1)};
318 const bool ok = sendCommand(static_cast<uint8_t>(ssd1331::Command::DrawRectangle), data,
319 sizeof(data));
320
321 if (filled) {
322 const uint8_t disable[2] = {0x00, 0x00};
323 sendCommand(static_cast<uint8_t>(ssd1331::Command::FillEnable), disable, 2);
324 }
325 return ok;
326}

339template <typename SpiType>
340bool SmartDisplay<SpiType>::ConfigureScroll(uint8_t h_offset, uint8_t start_row, uint8_t scroll_rows,
341 uint8_t v_offset, uint8_t interval_code,
342 bool activate) noexcept {
343 const uint8_t data[5] = {h_offset, start_row, scroll_rows, v_offset,
344 static_cast<uint8_t>(interval_code & 0x03)};
345 if (!sendCommand(static_cast<uint8_t>(ssd1331::Command::ScrollSetup), data, sizeof(data))) {
346 return false;
347 }
348 return sendCommandOnly(activate ? static_cast<uint8_t>(ssd1331::Command::ActivateScroll)
349 : static_cast<uint8_t>(ssd1331::Command::DeactivateScroll));
