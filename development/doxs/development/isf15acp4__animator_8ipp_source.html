40
41 for (uint8_t y = 0; y < h; ++y) {

57inline void Plasma(GraphicsContext& gfx, uint32_t frame, void* /*user_data*/) noexcept {
58 const uint8_t w = gfx.Width();
59 const uint8_t h = gfx.Height();
60 for (uint8_t y = 0; y < h; ++y) {
61 for (uint8_t x = 0; x < w; ++x) {
62 const float v = std::sin((x + frame) * 0.15f) + std::sin((y + frame) * 0.21f) +
