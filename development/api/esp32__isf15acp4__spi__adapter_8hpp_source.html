124 void GpioSet(isf15acp4::CtrlPin pin, isf15acp4::GpioSignal signal) noexcept {
125 const bool active = signal == isf15acp4::GpioSignal::Active;
126 switch (pin) {
127 case isf15acp4::CtrlPin::ChipSelect:
128 gpio_set_level(pins_.cs, active ? 0 : 1);
129 break;
130 case isf15acp4::CtrlPin::DataCommand:
131 gpio_set_level(pins_.dc, active ? 1 : 0);
132 break;
133 case isf15acp4::CtrlPin::Reset:
134 gpio_set_level(pins_.reset, active ? 0 : 1);
135 break;
136 case isf15acp4::CtrlPin::VccEnable:
137 gpio_set_level(pins_.vcc_enable, active ? 1 : 0);
138 break;
139 case isf15acp4::CtrlPin::Switch:
140 break;
141 }
142 }

144 bool GpioRead(isf15acp4::CtrlPin pin) noexcept {
145 if (pin != isf15acp4::CtrlPin::Switch || pins_.switch_in == GPIO_NUM_NC) {
146 return false;
147 }
148 const int level = gpio_get_level(pins_.switch_in);
149 return pins_.switch_active_low ? (level == 0) : (level == 1);
150 }
